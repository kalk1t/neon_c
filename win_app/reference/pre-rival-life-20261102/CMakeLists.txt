if(RR_BUILD_APP)
    if(RR_RAYLIB_SOURCE_DIR)
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
        add_subdirectory("${RR_RAYLIB_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/raylib")
        if(EMSCRIPTEN)
            # Emscripten EM_ASM integration requires GNU C extensions; core stays C17.
            set_target_properties(raylib PROPERTIES C_EXTENSIONS ON)
        endif()
        set(RR_RAYLIB_TARGET raylib)
    elseif(WIN32 AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(RR_RAYLIB_PACKAGE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/raylib-6.2-msvc-x64/raylib-6.2_win64_msvc16")
        if(NOT EXISTS "${RR_RAYLIB_PACKAGE}/lib/raylib.lib")
            message(FATAL_ERROR "Run scripts/Install-Raylib.ps1 before building.")
        endif()
        add_library(rr_raylib STATIC IMPORTED)
        set_target_properties(rr_raylib PROPERTIES
            IMPORTED_LOCATION "${RR_RAYLIB_PACKAGE}/lib/raylib.lib"
            INTERFACE_INCLUDE_DIRECTORIES "${RR_RAYLIB_PACKAGE}/include"
            INTERFACE_LINK_LIBRARIES "opengl32;gdi32;winmm;shell32;user32")
        set(RR_RAYLIB_TARGET rr_raylib)
    else()
        message(FATAL_ERROR "Supply RR_RAYLIB_SOURCE_DIR for this platform. Native binaries are Windows x64 MSVC only.")
    endif()

    add_executable(RoadstersRace src/app/main.c src/presentation/rr_scene.c)
    target_include_directories(RoadstersRace PRIVATE src/presentation)
    target_link_libraries(RoadstersRace PRIVATE rr_core ${RR_RAYLIB_TARGET})
    set_target_properties(RoadstersRace PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    if(MSVC)
        target_compile_options(RoadstersRace PRIVATE /W4 /WX)
    endif()
    if(EMSCRIPTEN)
        set_target_properties(RoadstersRace PROPERTIES SUFFIX ".html" C_EXTENSIONS ON)
        target_compile_definitions(RoadstersRace PRIVATE PLATFORM_WEB)
        target_link_options(RoadstersRace PRIVATE "-sUSE_GLFW=3" "-sALLOW_MEMORY_GROWTH=1" "-sMIN_WEBGL_VERSION=2" "-sMAX_WEBGL_VERSION=2" "-sFORCE_FILESYSTEM=1" "--preload-file" "${CMAKE_CURRENT_SOURCE_DIR}/assets@assets" "--shell-file" "${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html")
        # Stage the optimized JS as a fresh file in its output directory. On
        # Windows, compiler temporary files can otherwise retain private ACLs.
        add_custom_command(TARGET RoadstersRace POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_BINARY_DIR}/bin/RoadstersRace.js" "${CMAKE_BINARY_DIR}/bin/RoadstersRace.public.js"
            COMMAND ${CMAKE_COMMAND} -E rename "${CMAKE_BINARY_DIR}/bin/RoadstersRace.public.js" "${CMAKE_BINARY_DIR}/bin/RoadstersRace.js")
    else()
        add_custom_command(TARGET RoadstersRace POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:RoadstersRace>/assets")
        install(TARGETS RoadstersRace RUNTIME DESTINATION .)
        install(DIRECTORY assets DESTINATION .)
        install(DIRECTORY third_party/licenses DESTINATION .)
        install(FILES PLAYING.txt DESTINATION .)
    endif()
endif()
