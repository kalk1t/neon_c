cmake_minimum_required(VERSION 3.24)
project(RoadstersRaceRulesCheck LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(RR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
add_executable(rr_race_tests "${RR_ROOT}/tests/test_race.c" "${RR_ROOT}/src/core/rr_race.c")
target_include_directories(rr_race_tests PRIVATE "${RR_ROOT}/src/core")
if(MSVC)
    target_compile_options(rr_race_tests PRIVATE /W4 /WX /fp:precise)
else()
    target_compile_options(rr_race_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(rr_race_tests PRIVATE m)
endif()
add_test(NAME current_browser_race_rules COMMAND rr_race_tests)
add_library(rr_existing_core STATIC "${RR_ROOT}/src/core/rr_flight.c" "${RR_ROOT}/src/core/rr_track.c" "${RR_ROOT}/src/core/rr_recovery.c" "${RR_ROOT}/src/core/rr_corridor.c" "${RR_ROOT}/src/core/rr_city_plan.c")
target_include_directories(rr_existing_core PUBLIC "${RR_ROOT}/src/core")
if(MSVC)
    target_compile_options(rr_existing_core PRIVATE /W4 /WX /fp:precise)
else()
    target_compile_options(rr_existing_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(rr_existing_core PUBLIC m)
endif()
add_executable(rr_core_tests "${RR_ROOT}/tests/test_core.c")
target_link_libraries(rr_core_tests PRIVATE rr_existing_core)
add_test(NAME recorded_browser_flight_and_track COMMAND rr_core_tests)
add_executable(rr_render_policy_tests "${RR_ROOT}/tests/test_render_policy.c")
target_include_directories(rr_render_policy_tests PRIVATE "${RR_ROOT}/src/presentation")
target_link_libraries(rr_render_policy_tests PRIVATE rr_existing_core)
add_test(NAME bounded_render_policy COMMAND rr_render_policy_tests)
add_executable(rr_session_tests "${RR_ROOT}/tests/test_session.c" "${RR_ROOT}/src/core/rr_session.c" "${RR_ROOT}/src/core/rr_race.c")
target_link_libraries(rr_session_tests PRIVATE rr_existing_core)
if(MSVC)
    target_compile_options(rr_session_tests PRIVATE /W4 /WX /fp:precise)
else()
    target_compile_options(rr_session_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME race_session_lifecycle COMMAND rr_session_tests)

add_executable(rr_menu_policy_tests "${RR_ROOT}/tests/test_menu_policy.c")
target_include_directories(rr_menu_policy_tests PRIVATE "${RR_ROOT}/src/core")
add_test(NAME menu_activation_policy COMMAND rr_menu_policy_tests)

add_executable(rr_recovery_tests "${RR_ROOT}/tests/test_recovery.c")
target_link_libraries(rr_recovery_tests PRIVATE rr_existing_core)
if(MSVC)
    target_compile_options(rr_recovery_tests PRIVATE /W4 /WX)
endif()
add_test(NAME browser_route_recovery COMMAND rr_recovery_tests)

add_executable(rr_all_roads_tests "${RR_ROOT}/tests/test_all_roads.c" "${RR_ROOT}/src/core/rr_race.c")
target_link_libraries(rr_all_roads_tests PRIVATE rr_existing_core)
if(MSVC)
    target_compile_options(rr_all_roads_tests PRIVATE /W4 /WX)
endif()
add_test(NAME all_browser_road_geometry COMMAND rr_all_roads_tests)

add_executable(rr_road_picker_tests "${RR_ROOT}/tests/test_road_picker.c" "${RR_ROOT}/src/core/rr_road_picker.c" "${RR_ROOT}/src/core/rr_session.c" "${RR_ROOT}/src/core/rr_race.c")
target_link_libraries(rr_road_picker_tests PRIVATE rr_existing_core)
if(MSVC)
    target_compile_options(rr_road_picker_tests PRIVATE /W4 /WX)
endif()
add_test(NAME browser_road_picker COMMAND rr_road_picker_tests)

add_executable(rr_city_clearance_tests "${RR_ROOT}/tests/test_city_clearance.c")
target_link_libraries(rr_city_clearance_tests PRIVATE rr_existing_core)
if(MSVC)
    target_compile_options(rr_city_clearance_tests PRIVATE /W4 /WX)
endif()
add_test(NAME browser_city_clearance COMMAND rr_city_clearance_tests)
